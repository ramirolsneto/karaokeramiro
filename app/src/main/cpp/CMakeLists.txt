cmake_minimum_required(VERSION 3.22.1)
project(videoke_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LIB_DIR ${CMAKE_SOURCE_DIR}/../jniLibs/${ANDROID_ABI})
add_library(sherpa-c SHARED IMPORTED)
set_target_properties(sherpa-c PROPERTIES IMPORTED_LOCATION "${LIB_DIR}/libsherpa-onnx-c-api.so")
add_library(videoke SHARED separator.cpp)
target_include_directories(videoke PRIVATE ${CMAKE_SOURCE_DIR}/include)
find_library(log-lib log)
target_link_libraries(videoke sherpa-c ${log-lib})
